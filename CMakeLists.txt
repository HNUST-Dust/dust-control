cmake_minimum_required(VERSION 3.21)
project(dust_control VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "dust-control runtime requires Linux; build inside Orb Ubuntu or on the robot host")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BUILD_TESTING "Build automated regressions (never linked to production)" OFF)
option(DUST_ENABLE_FOXGLOVE "Build native read-only Foxglove telemetry" ON)
option(DUST_BUILD_MPC "Build the actively developed OSQP MPC algorithm" OFF)
option(DUST_BUILD_ETHERCAT "Build the actively developed EtherCAT backend" OFF)
option(DUST_BUILD_TOOLS "Build optional standalone commissioning/development apps" OFF)
option(DUST_ENABLE_IGH "Detect IgH EtherCAT support" ON)
option(DUST_ENABLE_SOEM "Detect SOEM EtherCAT support" ON)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
set(DUST_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${DUST_GENERATED_DIR}")
add_library(dust_headers INTERFACE)
target_include_directories(dust_headers INTERFACE "${PROJECT_SOURCE_DIR}/include")
include(cmake/Parameters.cmake)
add_subdirectory(modules/protocols)
add_subdirectory(modules/runtime)
add_subdirectory(modules/chassis)
add_subdirectory(modules/usb)
add_subdirectory(modules/telemetry)
if(DUST_BUILD_MPC)
  add_subdirectory(modules/mpc)
  add_subdirectory(modules/mpc_link)
endif()
if(DUST_BUILD_ETHERCAT)
  add_subdirectory(modules/ethercat)
endif()
add_subdirectory(apps)
include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
install(TARGETS chassis_controller RUNTIME DESTINATION bin)
install(DIRECTORY tools/foxglove/ DESTINATION share/dust-control/foxglove
        FILES_MATCHING PATTERN "*.json" PATTERN "*.md")
